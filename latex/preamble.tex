% \placeholder[height]{description}  —  grey box for missing figure
% SAFE: no \verb anywhere — uses only \texttt with explicit escaping
\newcommand{\placeholder}[2][7cm]{%
  \begin{center}
    \fcolorbox{gray!50}{gray!8}{%
      \parbox[c][#1][c]{0.84\linewidth}{%
        \centering\color{gray!70}\small\itshape #2%
      }%
    }%
  \end{center}%
}
